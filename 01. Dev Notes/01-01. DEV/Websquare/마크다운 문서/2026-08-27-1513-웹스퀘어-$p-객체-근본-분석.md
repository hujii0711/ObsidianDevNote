엔진 소스(`websquare/_websquare_/wqm_R1581_main.js`)를 직접 뜯어서 확인한 결과입니다.

## 결론부터

`$p`는 **전역 변수도, 엔진 API 객체도 아닙니다.** wframe(스코프 프레임)의 `scope` 객체에 꽂히는 **키 이름**이고, 그 값은 `WebSquare.uiplugin.wframe.scope` 인스턴스 — 즉 **자기 프레임의 uuid 하나만 들고 있는, 프레임 컨텍스트가 커링(currying)된 API 핸들**입니다.

```js
// 엔진 원문 (minify 해제)
wframe.scope = function(uuid) {
    this._parent_uuid = uuid;
    this._identifier  = "wframe_scope";
};
wframe.scope.prototype.getFrame = function() {
    return WebSquare.idCache[this._parent_uuid];
};
```

상태는 `_parent_uuid` 뿐입니다. 나머지 모든 메서드는 `getFrame()`으로 자기 프레임을 되찾아 그 컨텍스트를 붙여 전역 API로 위임합니다.

```js
scope.prototype.getComponentById = function(id) { return util.getComponentById(id, this.getFrameId()); };
scope.prototype.ajax  = function(o) { o.scope = this.getFrame().id; net.ajax(o); };
scope.prototype.openPopup = function(u,o,...) { o.scope_uuid = this._parent_uuid; util.openPopup(...); };
scope.prototype.emit  = function(ev) { return this.getFrame().trigger(ev, args); };
```

**`$p` = "나(이 페이지)를 아는 `$w`"** 가 한 줄 정의입니다.

## 1. 이름의 출처 — `$p`는 하드코딩이 아니라 설정값

```js
// WebSquare 기본값
noConflict:        "$w",
noConflictFunction:"$p",
```

`config.xml`의 `<noConflict value="" function=""/>`로 바꿀 수 있습니다. 즉 `$p`라는 글자 자체에 의미는 없고, **"페이지 스코프 API의 별칭"** 자리를 가리키는 슬롯 이름입니다.

`$w` = `WebSquare.api` (99개 멤버 — `ajax`, `openPopup`, `data`, `getComponentById`, `getParameter`, `top/parent/main`, `bigDecimal`, `executeSubmission` …).

## 2. `$p`는 두 종류다

### (a) 전역 `$p` — 프레임 스코프가 없을 때

```js
core._createGlobalVariable = function() {
    window["scwin"] = {};
    window["scwin"]["$w"] = WebSquare.api;   // noConflict
    window["scwin"]["$w"].id = "";           // ← 프리픽스 없음
    window["$p"] = window["scwin"]["$w"];    // noConflictFunction
};
```

`window.$p === window.scwin.$w === WebSquare.api`. 프레임을 모르므로 id 해석이 전역입니다.

### (b) 프레임 `$p` — 실제로 쓰는 것

```js
wframe.prototype.makeScopeVariable = function(lazy) {
    var idx = 0;
    for (var name of this.options.scopeVariable.split(",")) {   // "scwin"
        this._makeScopeVariable(name, WebSquare.noConflictFunction + (idx>0?idx:""), lazy);
        idx++;                                                   // "$p", "$p1", "$p2"...
    }
};

wframe.prototype._makeScopeVariable = function(varName, aliasKey, lazy) {
    this.scopeVariable[varName] = {};
    this.scope[varName]         = this.scopeVariable[varName];      // scope.scwin
    window[this.id + "_" + varName] = this.scopeVariable[varName];  // window.wTop_scwin (디버깅용 뒷문)

    if (this.options.scopeFunction) {                               // 기본 "$w"
        var n = this.scopeVariable[varName]["$w"] = new wframe.scope(this.uuid);

        for (var k in WebSquare.api)                                 // 전역 API 흡수
            if (k !== "data" && k !== "comp" && n[k] == null) n[k] = WebSquare.api[k];
        //  ↑ null 체크: prototype에 이미 있는 프레임 인지 버전이 이긴다

        n.data       = new wframe.scope.data(this.id);               // 프레임 바인딩 네임스페이스
        n.model      = new wframe.scope.model(this.id);
        n.bigDecimal = new wframe.scope.bigDecimal(this.id);
        n.id         = this.id + "_";                                // ★ 런타임 id 프리픽스

        this.scope[aliasKey] = this.scopeVariable[varName]["$w"];    // ★★ scope["$p"] = 그 객체
    }
};
```

여기서 세 가지가 결정됩니다:

| 사실 | 의미 |
|---|---|
| `scope["$p"] === scope.scwin.$w` | **`$p`와 `scwin.$w`는 동일 객체**입니다 |
| `n[k] == null` 일 때만 복사 | prototype의 프레임 인지 메서드가 전역 API를 **덮어씀** |
| `n.id = frameId + "_"` | `$p`가 곧 id 네임스페이스 |
| scopeVariable이 2개면 `$p1` | 두 번째 스코프 변수용 별칭이 자동 생성 |

**`$p.data` / `$p.model`은 별도 클래스**입니다 — `scope.data.prototype.get`은 `opts.scope_obj`에 자기 프레임 id를 자동 주입해서 `DataCollection.api.get`에 넘깁니다. `$p.data.get(...)`과 `$w.data.get(...)`이 다른 결과를 내는 이유입니다.

## 3. 그래서 `$p`가 페이지 스크립트에서 어떻게 "보이는가"

`$p`는 변수 선언된 적이 없습니다. **`with` 블록으로 주입**됩니다.

```js
util.scopingFuncBody = function(body, scopeComp, mode) {
    return "WebSquare.scope_obj=" + scopeComp + ".id;\n"
         + "with(" + scopeComp + ".scope){\n"          // ← scope 객체를 통째로 연다
         + "  var udc_obj = " + scopeComp + ".id;\n"
         + body
         + "\nWebSquare.scope_obj=null;\n}";
};
```

이 프로젝트의 wpack 컴파일 산출물이 정확히 그 모양입니다 — `_wpack_/webTop/cm/gcc/base/file.base.js`:

```js
{T:4, cdata: function(scopeObj){ with(scopeObj){ /* 원본 소스 그대로 */ } }}
```

그리고 `scope` 객체에는 `$p`, `scwin` 말고도 **하위 컴포넌트의 org_id가 전부 들어있습니다**:

```js
// 컴포넌트 생성 시
this.scope_obj.scope[this.org_id] = this;
```

→ 페이지에서 `btn_save.disable()`처럼 id를 맨 이름으로 쓸 수 있는 것도, `$p`가 보이는 것도 **완전히 같은 메커니즘**입니다. `$p`는 그 scope 테이블의 한 항목일 뿐입니다.

(참고: 사전 컴파일 경로에는 `with` 대신 구조분해 템플릿 `function(scopeObj){var {#_wq!$!@}=scopeObj; …}`가 있지만, 이 프로젝트 wpack 산출물은 `with` 방식입니다.)

## 4. 이 프로젝트에 직접 걸리는 사실들

### ① 최상위 페이지도 사실은 wframe이다

[config.xml:100](src/main/webapp/websquare/config.xml#L100) 의 `<strictMode id="wTop" value="true"/>` 때문에:

```js
if (WebSquare.strictMode && WebSquare.strictModeFrame == null) {
    var id = getConfiguration("/WebSquare/strictMode/@id");   // "wTop"
    WebSquare.strictModeFrame = controlFactory.create(id, {
        pluginName:"wframe", scope:true, scopeExternal:true, strictFrame:true, ...
    });
}
```

문서 전체가 `scope:true`인 가상 wframe **`wTop`** 으로 감싸집니다. 그리고 `scopeVariable`이 비어 있으면 [config.xml:92-97](src/main/webapp/websquare/config.xml#L92-L97)의 `<spa><variable value="scwin"/>`로 채워집니다.

**→ 이 프로젝트에서는 최상위 페이지의 `$p`조차 전역 `WebSquare.api`가 아니라 `wTop` 프레임의 scope 인스턴스입니다.** `$p.id === "wTop_"`이고, `$p.getComponentById()`는 `wTop` 스코프 안에서만 찾습니다.

게다가 strictMode는 id 탐색의 **스코프 체인 상승을 끊습니다**:

```js
util.getComponentById = function(id, scopeId) {
    if (!scopeId && WebSquare.strictModeFrame && WebSquare.strictModeFrame.id !== id)
        scopeId = WebSquare.strictModeFrame.id;
    if (scopeId) {
        var r = util.getComponentById(scopeId);
        for (var n = r; n != null && n.scope != null; n = n.scope_obj) {
            if (n.scope[id] != null) { i = n.scope[id].id; break; }
            if (n.scope_obj === n) break;
            if (WebSquare.strictMode) break;      // ★ 한 단계만 보고 중단
        }
    }
    return WebSquare.idCache[WebSquare.idToUUID[i]] || window[i];
};
```

즉 **`$p.getComponentById("someId")`는 부모 프레임으로 올라가지 않습니다.** 부모 것을 잡으려면 `$p.parent()`, `$p.top()`, `$p.main()`으로 스코프를 명시적으로 건너뛰어야 합니다 — [slip_gridView.xml:166-167](src/main/webapp/webTop/cm/exc/slip/slip_gridView.xml#L166-L167)의 주석이 이 동작을 두고 한 말입니다.

```js
scope.prototype.top    = function(){ return util.getTopWindow(); };
scope.prototype.parent = function(){ return this.getParentWindow(); };   // 부모 프레임의 scope
scope.prototype.self   = function(){ return this.getFrame().scope; };    // 내 scope 테이블 통째로
scope.prototype.main   = function(){ var f = this.getFrame()._getInheritFrame("api");
                                     return f ? f.scope : util.getMainFrame(); };
```

`parent()`/`self()`/`main()`이 반환하는 것은 **scope 객체**(= `{scwin, $p, 컴포넌트id들...}`)이지 `$p`가 아닙니다. 그래서 `$p.parent().scwin.xxx()`, `$p.top().$p.getComponentById(...)` 같은 체이닝이 나오는 겁니다.

### ② `$c.*` 공통 모듈에 `$p`를 첫 인자로 넘기는 이유

[config.xml:362-416](src/main/webapp/websquare/config.xml#L362-L416)의 `<projectCommon>` 모듈들은 페이지와 **완전히 다른 경로**로 로드됩니다:

```js
core._loadWebSquareCommon = function(type, promises, infos, rootVar) {
    ...
    var f = wframe._createEmptyFrame(name);       // ★ 화면 없는 빈 프레임을 만든다
    var m = f.scope;  m["scwin"] = {};
    for (script of scripts) script(m);            // with(scopeObj) 로 실행

    var S = {}, C = _parsePublicInfo(src);
    for (var k in m["scwin"])                     // publicInfo에 선언된 것만 추출
        if (_isPublicInfoVariable(C, m["scwin"], k, "scwin")) S[k] = m["scwin"][k];

    for (var q in m)
        if (q !== "scwin" && q !== WebSquare.noConflictFunction)   // ★★ "$p"를 명시적으로 제외
            if (_isPublicInfoComponent(q, m[q], C)) S[q] = m[q];

    window["$c"][name] = S;                       // → $c.file, $c.device ...
};
```

두 가지가 결정적입니다:

1. 공통 모듈 안의 `$p`는 **빈 프레임(`_createEmptyFrame`)의 `$p`** 입니다. 화면도 컴포넌트도 없으니 `getComponentById`가 아무것도 못 찾습니다.
2. export 객체 `S`를 만들 때 엔진이 **`$p`를 대놓고 빼버립니다** (`q !== noConflictFunction`).

그래서 `$c.file.readFile()`을 호출하면 그 함수의 클로저 `$p`는 쓸모없는 빈 프레임 것입니다. **호출자의 `$p`를 파라미터로 넘겨 클로저를 가리는 것 외에 방법이 없습니다.**

```js
// file.base.xml — 파라미터 $p 가 클로저의 $p 를 shadowing 한다
scwin.readFile = function($p, filePath) { ... };

// popup.base.xml — 호출부가 자기 $p 를 실어 보낸다
if (!$c.util.isEmpty($p, scwin.getCurrentBizScwin($p)._objCompLeastFocus)) { ... }
```

이 관례는 **필수**이고, 프로젝트가 임의로 정한 스타일이 아니라 엔진 구조가 강제한 것입니다.

**다만 주석 하나는 사실과 다릅니다.** [device.base.xml](src/main/webapp/webTop/cm/gcc/base/device.base.xml)·[file.base.xml](src/main/webapp/webTop/cm/gcc/base/file.base.xml) 곳곳의 `@param {Object} $p - WebSquare $p 객체 (프레임워크가 자동으로 전달)`는 이 설정에서 **성립하지 않습니다.** 자동 주입 경로는 딱 하나뿐인데:

```js
util.getGlobalFunction = function(expr, scopeId) {
    ...
    var g = getConfiguration("/WebSquare/wpack/common/@name");
    if (g && typeof g === "string") {
        var y = new RegExp("^(" + g.replace(/,/g,"|") + ")$");
        m = o.scope[WebSquare.noConflictFunction];             // 호출자의 $p
        for (f = 0; f < i.length-1; f++) if (y.test(i[f])) {
            var v = l.toString();
            if (v.slice(v.indexOf("(")+1, v.indexOf(")")).split(",")[0].trim()
                === WebSquare.noConflictFunction)               // 첫 파라미터 이름이 문자 그대로 "$p"
                l = l.bind(l, m);                               // ★ 여기서만 자동 바인딩
            break;
        }
    }
};
```

조건이 세 개입니다 — (a) `wpack/common/@name`에 네임스페이스가 등록돼 있고, (b) 호출이 **XML 이벤트 속성 문자열**(`ev:onclick="$c.file.readFile(...)"`)로 `getGlobalFunction`을 거치고, (c) 함수 첫 파라미터 이름이 정확히 `$p`일 것.

이 프로젝트는 [config.xml:10](src/main/webapp/websquare/config.xml#L10)이 `<common name="" value="true"/>` — **`@name`이 빈 문자열이라 `g`가 falsy이고, 자동 바인딩은 절대 발생하지 않습니다.** wpack 컴파일 산출물(`_wpack_/.../file.base.js`)을 확인해도 빌드 타임 `$p` 주입 흔적은 없고 `with(scopeObj)` 래핑만 있습니다. [cm.xml](src/main/webapp/webTop/cm/gcc/cm.xml)의 `_fwd` 헬퍼 주석("wpack이 `$p`를 주입한다")도 같은 오해에 서 있습니다 — 실제로는 JS 코드에서 직접 호출하는 한 항상 수동 전달이며, 지금 코드가 `slice(0)`으로 `$p`를 그대로 넘기고 있는 게 결과적으로 맞는 동작입니다.

### ③ EXC의 `$p`는 복사본 + `super`

[slip_textbox.xml:118](src/main/webapp/webTop/cm/exc/slip/slip_textbox.xml#L118)의 `$p.super("getFormatValue")`가 어디서 오는지:

```js
var S = n.options.scopeVariable.split(",")[0] || "scwin",
    C = n.options.scopeFunction;                     // "$w"
var q = {};
q[S] = {};
q[S][C] = WebSquare.extend({}, n.scope[S][C]);       // ★ 호스트 프레임 $p 의 얕은 복사본
q[WebSquare.noConflictFunction] = q[S][C];           // q["$p"]
q[compId] = w;

for (script of scripts) script(q);                   // EXC 스크립트를 이 scope 로 실행

var A = q[WebSquare.noConflictFunction];
if (typeof A.super !== "function") {
    A._excUUId = w.uuid;                             // ★ EXC 컴포넌트 자신의 uuid
    A.super = function(method, args) {
        var i = WebSquare.idCache[this._excUUId],
            r = i.getPluginName(),
            n = WebSquare.uiplugin[r];
        if (typeof n.prototype[method] !== "function") return;
        return n.prototype[method].apply(i, args);   // 원본 플러그인 구현 호출
    };
}
```

즉 EXC 안의 `$p`는 **호스트 프레임 `$p`의 얕은 복사본**에 `_excUUId` + `super`가 덧붙은 별개 객체입니다. `_parent_uuid`는 호스트 프레임을 가리키고, `_excUUId`는 EXC 컴포넌트 자신을 가리킵니다. `$p.super(name, argsArray)`는 상속 원본 플러그인의 prototype 메서드를 EXC 인스턴스에 `apply`합니다.

## 5. 정리

| 표현 | 실체 |
|---|---|
| `window.$w` | `WebSquare.api` (엔진 초기화 시 `window[noConflict] = api`) |
| `window.$p` | 전역 `scwin.$w` — 프레임 정보 없음 (`id === ""`) |
| 페이지 안의 `$p` | `frame.scope["$p"]` = `frame.scopeVariable.scwin.$w` = `new wframe.scope(frame.uuid)` |
| `$p === scwin.$w` | 같은 객체 (동일 참조) |
| `$p.id` | `프레임id + "_"` — 런타임 id 프리픽스 |
| `$p.getFrame()` | `WebSquare.idCache[$p._parent_uuid]` — 소속 wframe 컨트롤 |
| `$p.data` / `.model` | 프레임 id가 바인딩된 별도 클래스 인스턴스 |
| `$p.self()` | `frame.scope` (scope 테이블 자체) |
| `$p.parent()` / `.top()` / `.main()` | 다른 프레임의 **scope 객체** (`$p`가 아님) |
| `$p1`, `$p2` | `scopeVariable`이 2개 이상일 때의 추가 별칭 |
| `window.<frameId>_scwin` | 디버깅용 전역 뒷문 |
| `$w.debug.getScope(id).scwin.$w` | 콘솔에서 특정 프레임 `$p` 잡는 법 |

## 6. 실무 규칙

1. **컴포넌트 접근은 항상 `$p.getComponentById()`** — 전역 `window[id]`나 맨 id는 strictMode에서 프레임이 바뀌면 깨집니다.
2. **부모/최상위 접근은 명시적으로** — `$p.parent()`, `$p.top()`, `$p.main()`. strictMode는 스코프 체인을 자동으로 올라가지 않습니다.
3. **`$c.*` 공통 함수 호출 시 `$p`를 반드시 첫 인자로 전달** — 자동 주입은 이 설정에서 일어나지 않습니다.
4. **공통 모듈 내부에서 클로저 `$p`에 의존하지 말 것** — 빈 프레임의 것이라 무력합니다. 파라미터로 받은 `$p`만 사용하세요.
5. **`$p`를 지역 변수에 저장해 비동기 콜백으로 넘겨도 안전** — `_parent_uuid`만 붙들고 있으므로 프레임이 살아있는 한 유효합니다. 반대로 프레임이 `remove()`되면 `getFrame()`이 `undefined`를 반환하며 조용히 실패합니다(모든 메서드가 try/catch + `printStackTrace`로 감싸져 있어 **예외가 위로 전파되지 않습니다** — 디버깅 시 주의).
6. **`$p`와 `$w`를 섞어 쓰지 말 것** — `$w.data.get()`은 스코프 무시, `$p.data.get()`은 프레임 한정. 같은 이름의 dataList가 여러 프레임에 있으면 결과가 갈립니다.
