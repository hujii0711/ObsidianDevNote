```python
def node_with_config(state: State, config: RunnableConfig):
	"""config를 사용하는 노드: thread_id 등 설정 정보 접근 가능"""
	# config에서 thread_id 추출 (없으면 "unknown")
	thread_id = config.get("configurable", {}).get("thread_id", "unknown")
	print(f"Thread ID: {thread_id}")
	return {"results": f"안녕, {state['input']}!"}
```

4. ch03/ex01.py
```python
@tool
def get_user_info(runtime: ToolRuntime[UserContext]) -> str:
	"""Get information about the current user."""
	# Context에서 사용자 정보 가져오기
	user_id = runtime.context.user_id
	user_name = runtime.context.user_name
	user_email = runtime.context.user_email
	print(f"사용자 정보: ID={user_id}, 이름={user_name}, 이메일={user_email}")
	return f"User ID: {user_id}, Name: {user_name}, Email: {user_email}"
```
