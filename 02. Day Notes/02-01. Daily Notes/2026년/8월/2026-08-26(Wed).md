function aaa() {
	return Promise.resolve()
		.then(function () {
			console.log("1 시작");
			delay(1000, 1).then(v => console.log("1의 비동기 작업 완료:", v)); // return 없음!
			return 1; // 비동기 작업과 무관하게 즉시 1 반환
		})
		.then(function (tabId) {
			console.log("2 시작 (받은 값:", tabId, ")");
			return 2;
		})
		.then(function (tabId) {
			console.log("3 (받은 값:", tabId, ")");
			return 3;
		});
}

(async () => {
	const a = await aaa();
	console.log("a=======", a);
})();

```javascript
//네, 핵심은 "return 여부"입니다. `.then()` 콜백 내부에 비동기 코드가 있어도, 그 비동기 작업(Promise)을 return 하면 순차 실행이 보장됩니다. 하지만 return 하지 않으면 기다리지 않고 다음 단계로 넘어가 버립니다.

### 1) 비동기 코드를 `return` 한 경우 (순차 실행 
function delay(ms, value) {
	return new Promise(function (resolve) {
		setTimeout(function () {
			resolve(value);
		}, ms);
	});
}

function aaa() {
	return Promise.resolve()
		.then(function () {
			console.log("1 시작");
			return delay(1000, 1);   // 비동기 작업을 return
		})
		.then(function (tabId) {
			console.log("1 완료 후 받은 값:", tabId);
			console.log("2 시작");
			return delay(1000, 2);   // 비동기 작업을 return
		})
		.then(function (tabId) {
			console.log("2 완료 후 받은 값:", tabId);
			console.log("3");
			return 3;
		});
}

(async () => {
	console.time("전체시간");
	const a = await aaa();
	console.timeEnd("전체시간");
	console.log("a=======", a);
})();
```
